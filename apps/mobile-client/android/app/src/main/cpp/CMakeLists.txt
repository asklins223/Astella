cmake_minimum_required(VERSION 3.22.1)
project(astella_companion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(CUBISM "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native-sdk/CubismSdkForNative-5-r.5" ABSOLUTE)
set(FRAMEWORK_SOURCE OpenGL)
set(FRAMEWORK_DEFINITIOINS CSM_TARGET_ANDROID_ES2)
add_subdirectory("${CUBISM}/Framework" "${CMAKE_CURRENT_BINARY_DIR}/Framework")
target_compile_definitions(Framework PUBLIC CSM_TARGET_ANDROID_ES2)
target_include_directories(Framework PUBLIC "${CUBISM}/Core/include")
add_library(CubismCore STATIC IMPORTED)
set_target_properties(CubismCore PROPERTIES IMPORTED_LOCATION "${CUBISM}/Core/lib/android/${ANDROID_ABI}/libLive2DCubismCore.a")
add_library(astella_companion SHARED CompanionBridge.cpp)
target_include_directories(astella_companion PRIVATE "${CUBISM}/Samples/Metal/thirdParty/stb")
target_link_libraries(astella_companion Framework CubismCore GLESv2 log android)
